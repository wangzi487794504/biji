#### 项目实现白名单

* 常见的白名单实现方式

  * 文件配置
  * 数据库配置白名单表
  * 基于redis的set实现白名单表

* set的常用命令

  * scard key查看数量
  * sadd key val1 val2 增加
  * sismember key val 是否存在
  * smembers key 返回集合中的成员
  * srem key val 删除集合中元素

* redis实现白名单

  * 实现类

    ```java
    public class ArticleWriteServiceImpl implements ArticleWriteService {
        private static final String ARTICLE_WHITE_LIST = "auth_article_white_list";
        @Override
        public boolean authorInArticleWhiteList(Long authorId) {
            //是否存在
            return RedisClient.sIsMember(ARTICLE_WHITE_LIST,authorId);
        }
    
        @Override
        public void addAuthor2ArticleWhitList(Long userId) {
            //增加白名单
            RedisClient.sAdd(ARTICLE_WHITE_LIST,userId);
        }
    
        @Override
        public void removeAuthorFromArticleWhiteList(Long userId) {
            //删除白名单
            RedisClient.sRem(ARTICLE_WHITE_LIST,userId);
    
        }
    }
    ```

    

  * 配置类

    ```java
    @Configuration
    public class RedisConfig {
        public static final String REDIS_HOST = "127.0.0.1";
        @Bean
        public RedisTemplate<String,String> redisTemplate(){
            RedisConnectionFactory factory=new JedisConnectionFactory();
            RedisTemplate<String,String> redisTemplate = new RedisTemplate<>();
            redisTemplate.setConnectionFactory(factory);
            Jackson2JsonRedisSerializer jackson2JsonRedisSerializer=new Jackson2JsonRedisSerializer(Object.class);
            redisTemplate.setKeySerializer(new StringRedisSerializer());
            redisTemplate.setValueSerializer(jackson2JsonRedisSerializer);
            //设置序列化
            return redisTemplate;
        }
    }
    ```

  * client

    ```java
    @Component
    public class RedisClient {
        private static final Charset CODE = StandardCharsets.UTF_8;
        private static final String KEY_PREFIX = "pai_";
        @Autowired
        private static RedisTemplate<String, String> template;
        public static <T> boolean hSet(String key, String field,T ans) {
            return template.execute(new RedisCallback<Boolean>() {
                public Boolean doInRedis(RedisConnection connection) {
                    //在 Redis 操作的上下文中，RedisHashCommands 接口里的 hSet 方法用于操作 Redis 中的哈希（Hash）数据结构。
                    // hSet 方法用于将哈希表 key 中的字段 field 的值设置为 value。如果 key 不存在，会创建一个新的哈希表并进行设置；
                    // 如果 field 已经存在于哈希表中，它的值将被新的 value 覆盖。
                    return connection.hSet(keyBytes(key), valBytes(field), valBytes(ans));
                }
            });
        }
        public static byte[] keyBytes(String key) {
            nullCheck(key);
            key = KEY_PREFIX + key;
            return key.getBytes(CODE);
        }
        public static <T> byte[] valBytes(T val) {
            if (val instanceof String) {
                return ((String) val).getBytes(CODE);
            } else {
                return JsonUtil.toStr(val).getBytes(CODE);
            }
        }
        public static void nullCheck(Object... args) {
            for (Object obj : args) {
                if (obj == null) {
                    throw new IllegalArgumentException("redis argument can not be null!");
                }
            }
        }
        public static <T> Boolean sIsMember(String key, T value) {
            return template.execute(new RedisCallback<Boolean>() {
                @Override
                public Boolean doInRedis(RedisConnection connection) throws DataAccessException {
                    return connection.sIsMember(keyBytes(key), valBytes(value));
                }
            });
        }
        public static <T> Long sAdd(String key, T value) {
            return template.execute(new RedisCallback<Long>() {
                @Override
                public Long doInRedis(RedisConnection connection) throws DataAccessException {
                    return connection.sAdd(keyBytes(key), valBytes(value));
                }
            });
        }
        public static <T> Long sRem(String key, T value) {
            return template.execute(new RedisCallback<Long>() {
                @Override
                public Long doInRedis(RedisConnection connection) throws DataAccessException {
                    return connection.sRem(keyBytes(key), valBytes(value));
                }
            });
        }
        public static <T> Boolean hDel(String key, T value) {
            return template.execute(new RedisCallback<Boolean>() {
                @Override
                public Boolean doInRedis(RedisConnection connection) throws DataAccessException {
                    return connection.hDel(keyBytes(key), valBytes(value))>0;
                }
            });
        }
    
    }
    ```
