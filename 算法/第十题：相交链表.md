#### 相交链表

* 给你两个单链表的头节点 `headA` 和 `headB` ，请你找出并返回两个单链表相交的起始节点。如果两个链表不存在相交节点，返回 `null` 。

  ```java
  /**
   * Definition for singly-linked list.
   * public class ListNode {
   *     int val;
   *     ListNode next;
   *     ListNode(int x) {
   *         val = x;
   *         next = null;
   *     }
   * }
   */
  public class Solution {
      public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
          // 处理空指针异常
          if (headA == null || headB == null) {
              return null;
          }
          // 计算两个链表的长度，长的那个移动他们两个的差值
          int lengthA = 0;
          int lengthB = 0;
          ListNode tempA = headA;
          while (tempA != null) {
              lengthA++;
              tempA = tempA.next;
          }
          ListNode tempB = headB;
          while (tempB != null) {
              lengthB++;
              tempB = tempB.next;
          }
          // 让较长链表的指针先移动长度差值的步数
          for (int step = 0; step < (lengthA - lengthB > 0 ? lengthA - lengthB : lengthB - lengthA); step++) {
              if (lengthA - lengthB > 0) {
                  headA = headA.next;
              } else {
                  headB = headB.next;
              }
          }
          // 同步移动两个指针，直到找到相交节点或者遍历到链表末尾
          while (headA != null) {
              if (headA == headB) {
                  return headA;
              }
              headA = headA.next;
              headB = headB.next;
          }
          return null;
      }
  }
  ```
